def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()

def execute(argv,repo,env,stem,timeout,prompt=None):
 start=time.time();(stem.with_suffix('.command.json')).write_text(json.dumps(argv,indent=2));
 if prompt is not None:stem.with_suffix('.prompt.txt').write_text(prompt)
 with stem.with_suffix('.stdout').open('w') as out,stem.with_suffix('.stderr').open('w') as err:
  try:p=subprocess.run(argv,cwd=repo,env=env,input=prompt,text=True,stdout=out,stderr=err,timeout=timeout);rc=p.returncode
  except subprocess.TimeoutExpired:rc=124
 receipt={'returncode':rc,'start':start,'seconds':round(time.time()-start,3),'timeout_seconds':timeout};save(stem.with_suffix('.run.json'),receipt);return receipt,stem.with_suffix('.stdout').read_text()
def model(d,home,prompt,phase,model_id='gpt-6.1-sol',sid=None,readonly=False,timeout=600,profile=None,schema=False):
 output=d/(phase+'.last.txt');stem=d/phase
 cmd=['codex','exec']+(['resume'] if sid else [])+['--json','-m',model_id,'-c','model_reasoning_effort="low"','-c','approval_policy="never"']
 if sid:cmd+=['-c','sandbox_mode='+json.dumps('read-only' if readonly else 'danger-full-access')]
 else:cmd+=['-s','read-only' if readonly else 'danger-full-access','-C',str(d/'repo')]
 if schema:cmd+=['--output-schema',str(R/'review-schema.json')]
 cmd+=['-o',str(output)]+([sid] if sid else [])+['-']
 if profile:cmd=permissions.wrap_argv(cmd,profile)
 receipt,events=execute(cmd,d/'repo',env_for(home),stem,timeout,prompt)
 for line in events.splitlines():
  try:x=json.loads(line)
  except:continue
  if x.get('type')=='thread.started':sid=x['thread_id'];break
 if receipt['returncode']!=0:raise RuntimeError(f'{phase} exited {receipt["returncode"]}; retain partial session')
 if not sid or not output.exists():raise RuntimeError('missing finite session identity/output')
 return sid,output.read_text()
def packet(T,d,task):

 profile=d/'protected-checks.sb';profile.write_text(permissions.seatbelt_profile(permissions.load_permissions(T/'state'),primary_repo=repo,state_dir=T/'state',deny_write_subpaths=[str(check),str(manifest)]));assert permissions.wrapper_available()
 results=[]
 for i,cmd in enumerate(task['test_commands']+[json.loads(manifest.read_text())['checks'][0]['argv']]):
  p=subprocess.run(permissions.wrap_argv(cmd,profile),cwd=repo,env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1'),capture_output=True,text=True,timeout=300);log=d/f'protected-check-{i}.log';log.write_text(p.stdout+p.stderr);results.append({'command':cmd,'returncode':p.returncode,'status':'passed' if p.returncode==0 else 'failed','log_path':str(log),'output':p.stdout+p.stderr,'permission_wrapper':True})
 status=subprocess.check_output(['git','status','--porcelain','--untracked-files=all'],cwd=repo,text=True);changed=[l[3:] for l in status.splitlines()];sources={};diff=''
 for rel in task['allowed_paths']:
  p=repo/rel
  if p.is_file() and not p.is_symlink():
   text=p.read_text();sources[rel]={'sha256':sha(p),'text':text};diff+=''.join(difflib.unified_diff([],text.splitlines(True),fromfile='/dev/null',tofile='b/'+rel))
 p={'task':task,'sources':sources,'diff':diff,'tests':results,'findings':test_review.review_python_tests(repo,changed),'audit':{'status':status,'changed_paths':changed,'scope_clean':set(changed)==set(task['allowed_paths']),'head':subprocess.check_output(['git','rev-parse','HEAD'],cwd=repo,text=True).strip(),'base_tree':subprocess.check_output(['git','rev-parse','HEAD^{tree}'],cwd=repo,text=True).strip()},'screenshots':'not applicable; Python fixture','trusted_check_sha256':sha(check),'trusted_manifest':manifest.read_text(),'trusted_check_source':check.read_text()};path=d/'review-package.json';save(path,p);return path,p
